; Swords & Serpents — movement delta probe.
; Boots into the first dungeon, then injects dx/dy movement vars directly
; (G_0108/$0108 = dx, G_010C/$010C = dy) and measures $0325/$032D delta.
; G_017F/$017F = speed = 0x32 = 50 per the disasm; we test various magnitudes.
; Purpose: find the scale factor between G_0108/G_010C values and pixel position.
p 506A 0034
p 506B 0034
p 506C 0034
p 506D 0034
p 506E 0034
b 5072
r 8000000
p 018C 0001
g 7 56C9
n 5072
b 5072
r 8000000
r 8000000
r 8000000
m 0325 1
m 032D 1
; --- test 1: poke dx=+50 (G_017F value), dy=0  => expect X to increase
p 0108 0032
p 010C 0000
r 8000000
m 0325 1
m 032D 1
; --- settle (no input) for 1 frame to let game reset the vars
r 8000000
; --- test 2: poke dx=-50, dy=0 => expect X to decrease
p 0108 7FCE
p 010C 0000
r 8000000
m 0325 1
m 032D 1
; --- test 3: poke dx=0, dy=+50 => expect Y to increase
p 0108 0000
p 010C 0032
r 8000000
m 0325 1
m 032D 1
; --- test 4: poke dx=0, dy=-50 => expect Y to decrease
p 0108 0000
p 010C 7FCE
r 8000000
m 0325 1
m 032D 1
; --- test 5: poke dx=+10 => measure smaller delta
p 0108 000A
p 010C 0000
r 8000000
m 0325 1
m 032D 1
; --- test 6: poke dx=+1 => smallest step
p 0108 0001
p 010C 0000
r 8000000
m 0325 1
m 032D 1
q
